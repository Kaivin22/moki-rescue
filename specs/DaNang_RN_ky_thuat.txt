export function optimizeRoute(input: OptimizerInput): OptimizerResult {
  // BƯỚC 1: K-means clustering theo tọa độ GPS
  const clusters = kMeansCluster(input.places, input.numDays);

  // BƯỚC 2: Greedy Nearest Neighbor trong mỗi cluster
  const sortedClusters = clusters.map(cluster =>
    greedyNearestNeighbor(cluster, input.transport)
  );

  // BƯỚC 3: Constraint checking (giờ mở cửa)
  const withTime = sortedClusters.map(cluster =>
    applyTimeConstraints(cluster, input.startTime, input.endTime)
  );

  // BƯỚC 4: Budget check
  const budgetResult = checkBudget(withTime, input.budgetTotal);

  return buildResult(withTime, budgetResult);
}

export function recommendPlaces(
  places: Place[],
  profile: Profile,
  visited: string[]  // place IDs đã thăm
): Place[] {
  return places
    .filter(p => !visited.includes(p.id))
    .filter(p => isOpenThisMonth(p))
    .map(p => ({ ...p, score: calculateScore(p, profile) }))
    .sort((a, b) => b.score - a.score);
}

function calculateScore(place: Place, profile: Profile): number {
  const styleMatch = getStyleMatch(place.tags, profile.travel_style);
  const ratingScore = place.rating_avg / 5;
  const budgetScore = getBudgetScore(place, profile);
  const popularityScore = Math.min(place.rating_count / 100, 1);

  return styleMatch * 40
       + ratingScore * 25
       + budgetScore * 20
       + popularityScore * 15;
}

RAG Flow:
  1. Parse intent từ câu hỏi (tên địa điểm cụ thể hay chung?)
  2. Query Supabase lấy top 5 places liên quan
  3. Inject JSON vào system prompt
  4. Gọi Gemini → stream response
  5. Parse {{PLACE:tên}} → tạo tappable PlaceCard

System Prompt:
  VAI TRÒ: Hướng dẫn viên du lịch địa phương Đà Nẵng.
  ĐƯỢC PHÉP: Địa điểm, ẩm thực, di chuyển, thời tiết, văn hóa.
  KHÔNG ĐƯỢC: Y tế, pháp lý, tài chính.
  FORMAT: Dùng {{PLACE:tên}} khi đề cập địa điểm.
  DỮ LIỆU: [inject JSON từ Supabase]

react-native-maps tự động dùng:
  Android → Google Maps (cần GOOGLE_MAPS_API_KEY trong app.json)
  iOS     → Apple Maps (không cần key)

Cấu hình app.json:
  "android": {
    "config": {
      "googleMaps": { "apiKey": process.env.GOOGLE_MAPS_KEY }
    }
  }

Custom markers: dùng <Marker> với custom view theo category.
Cluster: react-native-maps-super-cluster (nếu cần).
Directions: mở native app qua Linking.openURL:
  iOS:     maps://maps.apple.com/?daddr=lat,lng
  Android: geo:lat,lng?q=destination

GOOGLE OAUTH
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Dùng @react-native-google-signin/google-signin + Supabase:
  1. GoogleSignin.configure({ webClientId: '...' })
  2. const { idToken } = await GoogleSignin.signIn()
  3. supabase.auth.signInWithIdToken({
       provider: 'google', token: idToken
     })

Trigger tự động tạo profile sau đăng ký (đã có trong DB).

PHASE 1 — Project Setup
  expo init danang-itinerary --template expo-template-blank-typescript
  Cài tất cả dependencies (package.json đầy đủ)
  Cấu hình app.json + eas.json + tsconfig.json
  Setup ESLint + Prettier
  Tạo cấu trúc thư mục đầy đủ

  .env + .env.example + .gitignore

  PlacePreviewBottomSheet
  Directions via Linking

PHASE 6 — Itinerary (QUAN TRỌNG NHẤT)

    BasicInfo → StyleBudget → SelectPlaces
    → Optimizing (loading) → Result
  app/itinerary/[id].tsx, edit.tsx, map.tsx

PHASE 7 — Share + Vote

PHASE 8 — AI Chat

PHASE 9 — VIP + Support

PHASE 10 — Admin + Editor
  Route guard (role check)
